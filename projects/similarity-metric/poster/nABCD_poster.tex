%---------------------------------------------------------------- LEFT COL ----
% LEFT.1  BACKGROUND
%------------------------------------------------------------------------------
\colsectionbanner{Background}

\posterbody
\begin{itemize}
  \item \textbf{MRCTs are the global standard;} pooling decisions hinge on
        effect modifier similarity. \textbf{ICH~E17} describes pooling
        qualitatively --- \textbf{no quantitative metric or threshold}.
  \item \textbf{Existing measures fall short:} SMD detects only location
        ($N(50,5^2)$ vs.\ $N(50,15^2)$ gives $\text{SMD}=0$); KS lacks
        clinical scale; KL is asymmetric and tail-unstable.
  \item \textbf{Goal.} A dissimilarity index capturing location, scale,
        and shape on a clinically interpretable scale.
\end{itemize}

%------------------------------------------------------------------------------
% LEFT.2  METHODS (with fig1 at \linewidth)
%------------------------------------------------------------------------------
\colsectionbanner{Methods}

\posterbody
\textbf{Definition.} The normalized Area Between CDFs Distance is the
Wasserstein-1 distance between region CDFs $F_1$ and $F_2$, normalized by a
robust pooled scale:
\[\boxed{\;\mathrm{nABCD}(F_1, F_2)
   \;=\; \dfrac{W_1(F_1, F_2)}{2 \cdot \IQRpool}\;}\]
where $W_1(F,G)=\int|F(x)-G(x)|\,dx$ is the area between the two CDFs and
$\IQRpool$ is the IQR of the pooled distribution. The factor of $2$
calibrates the scale so that a 1-IQR pure location shift yields exactly
$\mathrm{nABCD}=0.5$.

\textbf{Properties.} Symmetric, scale-free, sensitive to location/scale/shape
(unlike SMD, location only); baseline-only --- no outcome data needed.

\begin{center}
\includegraphics[width=0.80\linewidth]{fig1_nabcd_definition_color.pdf}\\[3pt]
\postercap{Shaded area between two CDFs equals $W_1(F_1,F_2)$;
nABCD divides by $2\cdot\IQRpool$.}
\end{center}

\posterbody
\textbf{Heterogeneity bound.} By Kantorovich--Rubinstein duality, if the CATE
$\tau(x)$ is $L$-Lipschitz, the between-region effect difference is bounded:
$|\bar\tau_1 - \bar\tau_2| \leq \Delta_{\max} = 2L\cdot\IQRpool\cdot\mathrm{nABCD}$.
This linkage from distributional metric to clinical scale is specific to
$W_1$ --- $W_2$ and KS admit no analog (Gibbs \& Su, 2002).

%------------------------------------------------------------------------------
% LEFT.3  CLINICAL CALIBRATION (placed after Methods per Tak's directive)
%------------------------------------------------------------------------------
\colsectionbanner{Clinical Calibration}

\posterbody
nABCD is a distance; to translate it into a clinical claim, two pathways
are offered depending on what is known about CATE sensitivity.
\begin{itemize}
  \item \textit{Pathway 1 ($L$ known):} $\Delta_{\max}$ is the worst-case
        regional treatment-effect difference on the clinical scale.
  \item \textit{Pathway 2 ($L$ unknown):} solve
        $L^{*} = \Delta_{\text{clin}} / (2\cdot\IQRpool\cdot\mathrm{nABCD})$;
        sponsor judges plausibility for the modifier and drug class.
\end{itemize}

\posterbody
\textbf{Scenarios.} Seven scenarios (S1--S7) isolate null, location-shift,

per scenario--$n$ cell, $n\in\{50,100,200\}$; bootstrap $B=2{,}000$,

%------------------------------------------------------------------------------
% RIGHT.2  APPLICATION: GUSTO-I (with fig3 at \linewidth)
%------------------------------------------------------------------------------
\colsectionbanner{Application: GUSTO-I}

\posterbody
\textbf{Setup.} Sponsor develops \textbf{Drug T} (thrombolytic for AMI)
and plans a Phase 3 MRCT. Planning-stage IPD: \textbf{GUSTO-I}
($N=40{,}830$; 16 anonymized regions). Anchor: \textbf{R8}
($n=2{,}916$); 15 candidate partners. Effect modifiers: \textbf{age}
and \textbf{systolic blood pressure}, flagged by Drug T's Phase 2.
Class-level CATE slopes unavailable $\Rightarrow$ $L^{*}$ pathway.

\vspace{2mm}

\begin{center}
\includegraphics[width=\linewidth]{fig3_gusto_r8_forest_color.pdf}\\[3pt]
\postercap{nABCD with 95\% percentile bootstrap CIs ($B=2{,}000$).
Age (A) and systolic blood pressure (B); partners ordered by ascending
nABCD.}
\end{center}

\vspace{2mm}

\posterbody
\textbf{Joint evaluation is essential.} R2 is 2nd-worst on age ($0.061$)
yet best on systolic blood pressure ($0.015$); R9 is nearly best on age
($0.017$) yet worst on systolic blood pressure ($0.110$). Single-modifier
rankings give opposite conclusions --- candidates must be evaluated
jointly.

\smallskip
\posterbody
\textbf{Eligibility \& leading candidate.} Plausible CATE-slope upper
bounds (FTT 1994):
$L_{\text{age}}^{\text{UB}} = 1{\times}10^{-2}/\text{yr}$,
$L_{\text{SBP}}^{\text{UB}} = 2{\times}10^{-3}/\text{mmHg}$. A region is
eligible if $L^{*} > L^{\text{UB}}$ at $\Delta_{\text{clin}}=1\%\text{pt}$
(30-day mortality). \textbf{Six regions are jointly eligible: R1, R4,
R5, R6, R14, R15.} \textbf{R4} is balanced across both modifiers
(3rd-lowest age $=0.016$; 4th-lowest systolic blood pressure $=0.042$)
and is selected as the leading pooling partner for R8.

\colsectionbanner{Discussion}

\subblockhead{Implications}
\posterbody
\begin{itemize}
  \item \textbf{Distributional + clinical scale.} The bound
        $|\bar\tau_1-\bar\tau_2|\leq 2L\cdot\IQRpool\cdot\mathrm{nABCD}$
        is unique to $W_1$; $W_2$ and KS admit no analog.
  \item \textbf{Beyond SMD.} Captures location, scale, and shape on a
        clinically interpretable scale; estimation-centered (report
        $\hnABCD$ + CI + $\Delta_{\max}$), not testing.
\end{itemize}

\subblockhead{Strengths}
\posterbody
\begin{itemize}
  \item \textbf{Fills the ICH~E17 gap} with transparent inputs
        supporting sponsor judgment, not a fixed cutoff.
  \item \textbf{Joint evaluation} of multiple effect modifiers
        prevents single-modifier rank reversals (R2 vs R9).
\end{itemize}

\subblockhead{Practical Recommendations}
\posterbody
\begin{itemize}
  \item \textbf{Plan with $L^{*}$} when class-level CATE slopes are
        unavailable; report $\Delta_{\max}$ when $L$ is known.
  \item \textbf{Sample size $n\geq 100$ per region}; sensitivity over
        plausible $L$; report all candidate modifiers, not best-case.
\end{itemize}
