\begin{block}{1.\ Motivation}
\textbf{Multi-regional clinical trials (MRCTs)} are the standard paradigm for
global drug development. ICH~E17 describes pooling decisions in terms of
\emph{similarity of effect modifier distributions} but gives
\textbf{no quantitative metric, threshold, or procedure}.

\medskip
\textbf{Why it matters.} With CATE $\tau(x)=E[Y(1)-Y(0)\mid X=x]$,
\[\bar\tau_r = \int \tau(x)\,dF_r(x),\]
so regions with different effect modifier distributions can observe different
treatment effects even when the drug works identically at the individual
level.

\medskip
\textbf{Gap.} The standardized mean difference (SMD) only detects
\emph{location} shifts. $N(50,5^2)$ vs.\ $N(50,15^2)$ yields SMD$=0$ despite a
3-fold difference in spread; scale and shape differences are invisible.

\medskip
\textbf{Objective.} A distributional dissimilarity index that
(i)~captures features beyond location, (ii)~has a clinically interpretable
scale, and (iii)~carries a theoretical link to treatment-effect
heterogeneity.
\end{block}

\begin{block}{2.\ The nABCD Dissimilarity Index}
Wasserstein-1 distance between CDFs:
\[W_1(F, G) = \int_{-\infty}^{\infty}\!|F(x)-G(x)|\,dx,\]
normalized by a robust scale:
\[\boxed{\;
\mathrm{nABCD}(F_1, F_2) \;=\; \frac{W_1(F_1, F_2)}{2 \cdot \mathrm{IQR}_{\text{pooled}}}
\;}\]
where $\mathrm{IQR}_{\text{pooled}}$ is computed from
$F_{\text{pool}}=\lambda F_1+(1-\lambda)F_2$, $\lambda=n_1/(n_1+n_2)$.
The factor of 2 calibrates so that a 1-IQR location shift yields
$\mathrm{nABCD}=0.5$.

\medskip
\begin{center}
\includegraphics[width=0.85\linewidth]{fig2_nabcd_definition.pdf}\\[2pt]
{\small\textit{Shaded area between two CDFs $= W_1(F_1,F_2)$.}}
\end{center}

\medskip
\textbf{Properties.}
\begin{itemize}
  \item $\geq 0$; $=0 \iff F_1=F_2$; symmetric; scale-free.
  \item Responds to location, scale, skewness, and shape.
  \item Requires only \emph{baseline} effect modifier data --
        from prior trials, registries, or RWE.
\end{itemize}
\end{block}

\begin{block}{3.\ Heterogeneity Bound \& Clinical Calibration}
By Kantorovich--Rubinstein duality,
\[W_1(F_1,F_2)=\sup_{\|f\|_{\mathrm{Lip}}\leq 1}
  \left|\int f\,dF_1-\int f\,dF_2\right|.\]
If $\tau(x)$ is Lipschitz with constant $L$, then $\tau/L$ is 1-Lipschitz and
\[|\bar\tau_1 - \bar\tau_2| \;\leq\; L\cdot W_1(F_1, F_2).\]
Substituting gives the \textbf{clinical heterogeneity bound}:
\[\boxed{\;\Delta_{\max}
   = 2\,L \cdot \mathrm{IQR}_{\text{pooled}} \cdot \mathrm{nABCD}(F_1, F_2)\;}\]
with bootstrap CI by linear mapping of the nABCD percentile interval.
$L$ is the per-unit CATE slope from Phase~2 interaction analysis or
subgroup-specific meta-analyses.

\medskip
\textit{A $W_1$-specific property:} $W_2$ and KS admit no analogous
heterogeneity bound (Gibbs \& Su, 2002).
\end{block}

\begin{block}{4.\ $L^{*}$ Reverse Calculation (unknown $L$)}
At the planning stage, $L$ is rarely numerically available. Rearranging the
heterogeneity bound:
\[\boxed{\;L^{*} \;=\; \dfrac{\Delta_{\text{clin}}}
  {2 \cdot \mathrm{IQR}_{\text{pooled}} \cdot \mathrm{nABCD}}\;}\]
$L^{*}$ is the \textbf{CATE sensitivity required} for the observed
distributional difference to yield a clinically meaningful
$\Delta_{\text{clin}}$.

\medskip
\textbf{Decision workflow.}
\begin{enumerate}
  \item Rank candidate partners by $\hnABCD$ with bootstrap CIs.
  \item Report $L^{*}$ at pre-specified $\Delta_{\text{clin}}$.
  \item Sponsor judges plausibility of $L^{*}$ for the effect modifier and
        drug class.
\end{enumerate}
This \emph{estimation-centered} framework deliberately avoids binary
accept/reject testing and fixed nABCD cutoffs.
\end{block}

\begin{block}{5.\ Implementation \& Data Requirements}
\textbf{Inference.} Percentile bootstrap ($B \geq 2{,}000$) for $\hnABCD$ CIs;
$L^{*}$ CI obtained by the same linear percentile mapping. BCa showed inferior
coverage for this bounded statistic. Recommended $n \geq 100$ per region.

\medskip
\textbf{Data source flexibility.} nABCD requires only \emph{baseline} effect
modifier distributions -- not treatment outcomes. Regional data may be drawn
from prior trials, disease registries, electronic health records, or other
real-world evidence (RWE) sources, aligning with ICH~E6(R3). Representativeness
of each region's data with respect to the target trial population is the sole
prerequisite.

\medskip
\textbf{Bridge to validation.} The next column quantifies how reliably
$\hnABCD$ recovers the true index and how it compares with SMD across
systematic distributional alterations.

\medskip
\textbf{Software.} Reference R implementation available; an R package with
bootstrap inference and clinical calibration is under development.
\end{block}

\begin{itemize}
  \item True $\mathrm{nABCD}\geq 0.1$: bias $<0.02$, coverage $0.88$--$0.96$

  \item \textbf{Percentile} bootstrap preferred over BCa (boundary).
  \item Recommended: $n\geq 100$ per region.
\end{itemize}

\begin{block}{9.\ Framework Philosophy}
\begin{itemize}
  \item \textbf{Estimation-centered, not testing.} Report $\hnABCD$, bootstrap
        CI, and $\Delta_{\max}$ or $L^{*}$ -- never a binary verdict.
  \item \textbf{No universal cutoff.} Cohen's-$d$-style benchmarks are
        context-free; nABCD's clinical weight depends on $L$, which is
        external to the distributional comparison.
  \item \textbf{Ranking + calibration.} Rank partners by $\hnABCD$; translate
        to clinical scale via $\Delta_{\max}$ ($L$ known) or $L^{*}$
        ($L$ unknown).
  \item \textbf{Two-pathway calibration.} Same framework for confirmatory
        (Phase~2 $L$) and planning (sensitivity via $L^{*}$) settings.
  \item \textbf{ICH~E17 aligned.} ``Similar enough'' remains with sponsor and
        regulators, informed by transparent quantitative inputs.
\end{itemize}

\medskip
\textbf{Limitations.} Continuous effect modifiers only; univariate
(multivariate OT extension is future work); near-boundary positive bias for
true $\mathrm{nABCD}\lesssim 0.05$; requires representative baseline data.
\end{block}

\begin{block}{10.\ Case Study: Hypothetical Drug T MRCT}
A sponsor develops \textbf{Drug T}, a novel thrombolytic for acute myocardial
infarction, and plans a Phase~3 MRCT. At the planning stage the sponsor uses
IPD from \textbf{GUSTO-I} ($N=40{,}830$; 16 anonymized regions) to assess
distributional similarity of candidate effect modifiers across regions and to
identify pooling partners for a designated anchor region.

\medskip
\textbf{Candidate effect modifiers.} Age and systolic blood pressure (SBP),
flagged by Drug~T's Phase~2 exploratory analyses; class-level numerical CATE
slopes are unavailable (FTT meta-analysis reports benefit ``irrespective of
age'' without per-year slope). $L$ is therefore treated as unknown and the
$L^{*}$ pathway applies to both.

\medskip
\textbf{Region 8 as anchor} ($n=2{,}916$).

\begin{block}{13.\ Conclusions}
\begin{enumerate}
  \item \textbf{nABCD} -- a normalized Wasserstein-1 index -- captures full
        distributional differences (location, scale, shape, skew) that SMD
        misses by construction.
  \item The heterogeneity bound
        $|\bar\tau_1-\bar\tau_2|\leq 2L\cdot\mathrm{IQR}_{\text{pooled}}\cdot\mathrm{nABCD}$
        translates distributional distance into a clinical-scale bound --
        a property unique to $W_1$.
  \item Seven-scenario simulation shows satisfactory percentile bootstrap
        performance (bias $<0.02$; coverage $0.88$--$0.96$) for true
        $\mathrm{nABCD}\geq 0.1$ at $n\geq 100$.
  \item The GUSTO-I illustration shows the framework in action: ranking,
        CIs, and $L^{*}$ identify R4/R6/R13 as leading pooling candidates for
        Region~8, while highlighting the R2 vs.\ R9 asymmetry that
        single-modifier rankings would obscure.
  \item Fills a methodological gap in ICH~E17 implementation by supplying
        \emph{transparent quantitative inputs} without imposing a universal
        cutoff.
\end{enumerate}

\medskip
\textbf{Available.} R code for nABCD and bootstrap inference; an R package is
under development.
\end{block}
